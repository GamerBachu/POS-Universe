Yes, you can absolutely remove any. In a professional TypeScript environment, we want to replace any with a Generic Type Variable (usually denoted as T). This tells TypeScript: "The data can be anything, but once it's defined, it must stay consistent."

Here is how you remove any from your form state and the useActionState hook.

1. The Generic Interface
By replacing any with T, you make the interface reusable for every form in your app.

TypeScript
// @/types/form.ts
export interface IFormState<T = unknown> {
  success: boolean;
  message: string | null;
  errors: Record<string, string[]> | null;
  data?: T; // Holds the typed object (e.g., IUserProfile)
}
2. The Type-Safe Action
Now, when you define your action, you explicitly tell it what type of data it returns.

TypeScript
import { IFormState } from "@/types/form";
import { IUserProfile } from "@/types/user";

export async function updateProfileAction(
  prevState: IFormState<IUserProfile>, 
  formData: FormData
): Promise<IFormState<IUserProfile>> {
  
  // Logic here...
  return {
    success: true,
    message: "Updated",
    errors: null,
    data: { /* ... updated profile data ... */ }
  };
}
3. Usage with useActionState
When you call the hook, TypeScript will now infer (automatically detect) the types based on your action function. You no longer need to manually define types inside the component.

TypeScript
export function ProfileForm() {
  // TypeScript knows 'state' is IFormState<IUserProfile> 
  // because updateProfileAction is typed.
  const [state, action, isPending] = useActionState(updateProfileAction, {
    success: false,
    message: null,
    errors: null
  });

  return (
    <form action={action}>
       <TextBoxWithLabel 
         label="Phone"
         name="phoneNumber"
         error={state.errors?.phoneNumber?.[0]} // Perfectly typed!
       />
       {/* ... */}
    </form>
  );
}

async function updateProfileAction(prevState: any, formData: FormData) {

export function ProfileUpdateForm() {
  // state: result from action, action: function to trigger, isPending: loading state
  const [state, action, isPending] = useActionState(updateProfileAction, { error: null });

  return (
    <form action={action} className="space-y-6">
      <TextBoxWithLabel

      >
        {isPending ? "Saving..." : "Save Profile"}
      </button>
    </form>
  );
}

export function QuickUserForm() {
  const [formData, setFormData] = useState({
    firstName: "",
    lastName: ""
  });

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const { name, value } = e.target;
    
    setFormData(prev => ({
      ...prev,
      [name]: value // Dynamic key update
    }));
  };

  return (
    <div className="space-y-4">
      <TextBoxWithLabel 
        label="First Name"
        name="firstName" // Important: matches state key
        value={formData.firstName}
        onChange={handleInputChange}
      />
      <TextBoxWithLabel 
        label="Last Name"
        name="lastName" // Important: matches state key
        value={formData.lastName}
        onChange={handleInputChange}
      />
    </div>
  );
}
